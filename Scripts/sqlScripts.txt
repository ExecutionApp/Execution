create or replace FUNCTION "DAILY_PER_PARTY_FEES"(
    RES_PARTY IN NUMBER)
  RETURN NUMBER
IS
  SUM_FEES NUMBER DEFAULT 0 ;
  FLAG     VARCHAR2(3) DEFAULT 2 ;
BEGIN
  DBMS_OUTPUT.PUT_LINE('SUM_FEES = ' || SUM_FEES);
  SELECT DECISIONS.work_amount_flag
  INTO FLAG
  FROM EXC_COURT_BC_CHARGES_PARTIES PARTIES,
    EXC_COURT_BC_CHARGE_DECISIONS DECISIONS
  WHERE PARTIES.CHARGE_DECISION_ID = DECISIONS.ID
  AND PARTIES.RES_PARTY_ID         = RES_PARTY
  AND ROWNUM                       = 1 ;
  IF (FLAG                         = 'Y' ) THEN
    SELECT NVL (SUM(FEES.FEES_AMOUNT), 0)
    INTO SUM_FEES
    FROM EXC_COURT_BC_CHARGES_PARTIES PARTIES,
      EXC_COURT_BC_DECISION DECISION,
      EXC_COURT_BC_DECISION_FEES FEES
    WHERE DECISION.ID        = FEES.DECISION_ID (+)
    AND PARTIES.DECISION_ID  = DECISION.ID
    AND PARTIES.RES_PARTY_ID = RES_PARTY
    AND ROWNUM               = 1
    AND FEES.FEES_TYPE_ID    = 2;
  END IF ;
DBMS_OUTPUT.PUT_LINE('SUM_FEES = ' || SUM_FEES);
RETURN SUM_FEES;
EXCEPTION
WHEN NO_DATA_FOUND THEN
  RETURN 0;
END;

create or replace FUNCTION "PENALTY_PER_PARTY_FEES"(
    RES_PARTY IN NUMBER)
  RETURN NUMBER
IS
  SUM_FEES NUMBER DEFAULT 0 ;
  FLAG     NUMBER DEFAULT 2 ;
BEGIN
  DBMS_OUTPUT.PUT_LINE('SUM_FEES = ' || SUM_FEES);
  SELECT PARTIES.SANCTION_COPIED_FLAG
  INTO FLAG
  FROM EXC_COURT_BC_CHARGES_PARTIES PARTIES
  WHERE PARTIES.RES_PARTY_ID = RES_PARTY
  AND ROWNUM                 = 1 ;
  IF (FLAG                   = 0) THEN
    SELECT NVL (SUM(FEES.FEES_AMOUNT), 0) 
    INTO SUM_FEES
    FROM EXC_COURT_BC_CHARGES_PARTIES PARTIES,
      EXC_COURT_BC_DECISION DECISION,
      EXC_COURT_BC_DECISION_FEES FEES
    WHERE DECISION.ID                = FEES.DECISION_ID (+)
    AND PARTIES.DECISION_ID          = DECISION.ID
    AND PARTIES.RES_PARTY_ID         = RES_PARTY
    AND PARTIES.SANCTION_COPIED_FLAG = 0 
    AND FEES.FEES_TYPE_ID = 2;
  END IF ;
IF (FLAG = 1 ) THEN
  SELECT NVL (SUM(FEES.FEES_AMOUNT), 0) 
  INTO SUM_FEES
  FROM EXC_COURT_BC_CHARGES_PARTIES PARTIES,
    EXC_COURT_BC_DECISION DECISION,
    EXC_COURT_BC_DECISION_FEES FEES
  WHERE DECISION.ID                = FEES.DECISION_ID (+)
  AND PARTIES.DECISION_ID          = DECISION.ID
  AND PARTIES.RES_PARTY_ID         = RES_PARTY
  AND ROWNUM                       = 1
  AND PARTIES.SANCTION_COPIED_FLAG = 1 
  AND FEES.FEES_TYPE_ID = 2;
END IF ;
DBMS_OUTPUT.PUT_LINE('SUM_FEES = ' || SUM_FEES);
RETURN SUM_FEES;
EXCEPTION
WHEN NO_DATA_FOUND THEN
  RETURN 0;
END;

CREATE OR REPLACE FUNCTION FIX_STRING (P_STRING VARCHAR2)
   RETURN VARCHAR2
IS
    V_STRING VARCHAR2(1000);
BEGIN
    V_STRING := REPLACE(P_STRING,'أ','ا');
    V_STRING := REPLACE(V_STRING,'إ','ا');
    V_STRING := REPLACE(V_STRING,'ة','ه');
    V_STRING := REPLACE(V_STRING,'ؤ','و');
    V_STRING := REPLACE(V_STRING,'ى','ي');
    V_STRING := REPLACE(V_STRING,'عبد ','عبد');
    V_STRING := REPLACE(V_STRING,'ابو ','ابو');
    V_STRING := REPLACE(V_STRING,'أبو ','ابو');
    V_STRING := REPLACE(V_STRING,'أبو','ابو');
    RETURN V_STRING;
END;
